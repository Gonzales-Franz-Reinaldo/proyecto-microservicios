¿Qué es Angular?

📱 Framework TypeScript

Framework de código abierto desarrollado y mantenido por Google para construir aplicaciones web dinámicas.

🎯 Single Page Application

Permite crear SPAs con una experiencia de usuario fluida sin recargas de página.

🔧 Basado en Componentes

Arquitectura modular donde cada componente encapsula su HTML, CSS y lógica TypeScript.

⚡ Alto Rendimiento

Optimización automática, lazy loading, y compilación AOT para aplicaciones rápidas.

🌐 Servicios y HTTP Client

import { HttpClient } from '@angular/common/http'; import { Injectable, inject } from '@angular/core'; @Injectable({ providedIn: 'root' }) export class EventosService { private http = inject(HttpClient); private apiUrl = 'http://localhost:3001/eventos'; getAll() { return this.http.get<Evento[]>(this.apiUrl); } getById(id: number) { return this.http.get<Evento>(`${this.apiUrl}/${id}`); } create(evento: EventoCreate) { return this.http.post<Evento>(this.apiUrl, evento); } }

🔒 HTTP Interceptors

import { HttpInterceptorFn } from '@angular/common/http'; export const authInterceptor: HttpInterceptorFn = (req, next) => { const token = localStorage.getItem('access_token'); // Lista de APIs que requieren token const apiUrls = [ 'http://localhost:3000', // Usuarios 'http://localhost:3001', // Eventos 'http://localhost:3002' // Compras ]; const isApiRequest = apiUrls.some(url => req.url.startsWith(url)); if (token && isApiRequest) { const cloned = req.clone({ setHeaders: { Authorization: `Bearer ${token}` } }); return next(cloned); } return next(req); };
